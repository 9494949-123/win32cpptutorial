📌 前言与环境说明

本教程适配有C++基础的进阶学习者，同时附赠完整小白零基础入门章节，兼顾原理深度与实操落地。Win32 API是Windows系统原生GUI开发接口，无需安装任何第三方库，纯净实现桌面窗口、控件、交互功能。

适配环境

  • 操作系统：Windows 全版本
  • 编译器：MinGW-w64 / MSVC

前置基础

  • C++ 基础语法、指针、结构体
  • 理解回调函数、消息驱动思想

核心区分：A后缀函数（ANSI char*）、W后缀函数（Unicode WCHAR*，现代Windows推荐），无后缀宏自动适配字符集。

1. 核心概念总览

  • HINSTANCE：程序实例句柄，唯一标识当前进程模块。
  • HWND：窗口句柄，所有窗口、按钮、输入框的唯一标识，属于系统内核对象，并非C++指针。
  • WndProc 窗口过程：系统回调函数，用户不主动调用，Windows推送所有界面事件到此函数处理。
  • MSG 消息结构体：存储系统事件信息，包含消息ID、附加参数 wParam/lParam。

消息循环：GUI程序核心常驻逻辑，三步执行：

  1. GetMessage：阻塞等待用户操作，无事件则休眠（低CPU占用）
  2. TranslateMessage：翻译键盘原始按键为字符消息
  3. DispatchMessage：分发消息到对应窗口的WndProc
  • 程序入口差异：控制台程序 main()，Win32 GUI程序 WinMain()
  • 编译关键：MinGW编译Win32 GUI程序必须加链接参数 -luser32；子系统 WINDOWS 无黑框，CONSOLE 附带控制台。

2. 最小完整模板（Unicode W版本·推荐）

现代Windows原生支持Unicode，优先使用W后缀API，无乱码问题，兼容性最强。

#include <windows.h>

// 窗口过程回调
LRESULT CALLBACK WndProc(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam)
{
    switch(uMsg)
    {
    case WM_CREATE:
        // 窗口刚创建完成时触发，适合在这里创建子控件（按钮、输入框）
        break;

    case WM_DESTROY:
        // 用户点击关闭按钮，窗口销毁，发送退出消息跳出消息循环
        PostQuitMessage(0);
        break;

    default:
        // 没有处理的消息必须交给默认窗口过程，否则窗口直接失效
        return DefWindowProcW(hWnd, uMsg, wParam, lParam);
    }
    return 0;
}

int WINAPI WinMain(HINSTANCE hInst, HINSTANCE hPrev, LPWSTR lpCmd, int nShow)
{
    // 1. 填充窗口类结构体 WNDCLASSEXW
    WNDCLASSEXW wc = {0};
    wc.cbSize        = sizeof(WNDCLASSEXW);
    wc.lpfnWndProc   = WndProc;        //绑定消息回调
    wc.hInstance     = hInst;
    wc.lpszClassName = L"MyWin32Demo"; //窗口类名，宽字符串 L""
    wc.hCursor       = LoadCursorW(NULL, IDC_ARROW);
    wc.hbrBackground = (HBRUSH)(COLOR_WINDOW + 1); //窗口背景色

    //注册窗口类到操作系统
    if(!RegisterClassExW(&wc))
    {
        MessageBoxW(NULL, L"注册窗口类失败",L"Error", MB_ICONERROR);
        return 0;
    }

    //2. 创建主窗口 CreateWindowExW，扩展创建函数（比CreateWindow功能更强）
    HWND hMainWnd = CreateWindowExW(
       0,
        L"MyWin32Demo",          //窗口类名，必须和注册的一致
        L"Win32 API Demo",       //窗口标题
        WS_OVERLAPPEDWINDOW,     //窗口样式：标题栏、最小化最大化关闭
        CW_USEDEFAULT, CW_USEDEFAULT, //x,y 位置交给系统默认
        600, 400,                //宽、高
        NULL, NULL, hInst, NULL );

    if(!hMainWnd)
    {
        MessageBoxW(NULL, L"创建主窗口失败", L"Error", MB_ICONERROR);
        return 0;
    }

    ShowWindow(hMainWnd, nShow);
    UpdateWindow(hMainWnd); //触发WM_PAINT重绘

    //3. 消息循环
    MSG msg;
    while(GetMessageW(&msg, NULL, 0, 0) > 0)
    {
        TranslateMessage(&msg);
        DispatchMessageW(&msg);
    }
    return (int)msg.wParam;
}

MinGW 编译命令：

g++ main.cpp -o win32demo.exe -luser32

代码流程拆解

  1. 初始化窗口结构体，绑定消息回调、样式、实例句柄
  2. 向系统注册自定义窗口类
  3. 基于注册类创建可视化主窗口
  4. 显示并刷新窗口，开启常驻消息循环

3. 创建标准子控件（按钮 BUTTON、编辑框 EDIT）

系统内置控件无需手动注册窗口类，直接通过 CreateWindowExW 创建，所有控件本质都是子窗口。

控件必备属性

  • WS_CHILD：标记为子控件，依附主窗口
  • WS_VISIBLE：创建后直接显示控件
  • 绑定父窗口HWND、唯一整数控件ID

控件统一在 WM_CREATE 消息中创建（窗口初始化阶段，无闪烁、无报错）：

case WM_CREATE:
{
   HWND hBtn = CreateWindowExW(
       0,
        L"BUTTON",
        L"点击我",
        WS_CHILD | WS_VISIBLE | BS_PUSHBUTTON,
        20,20,120,40,  // x y width height
        hWnd,
        (HMENU)1001, //控件ID 1001
        ((LPCREATESTRUCTW*)lParam)->hInstance,
        NULL
    );

    HWND hEdit = CreateWindowExW(
        0,
        L"EDIT",
        L"在这里输入",
        WS_CHILD | WS_VISIBLE | WS_BORDER,
       160, 20, 220,30,
        hWnd,
        (HMENU)1002, //编辑框ID
        ((LPCREATESTRUCTW*)lParam)->hInstance,
        NULL
    );
    break;
}

按钮点击事件捕获 WM_COMMAND

按钮点击触发 WM_COMMAND 消息，通过 wParam 解析控件ID和点击事件：

case WM_COMMAND:
{
    UINT notify = HIWORD(wParam);
    UINT cid    = LOWORD(wParam);
    if(cid == 1001 && notify == BN_CLICKED)
    {
        MessageBoxW(hWnd, L"按钮被点击！", L"提示", MB_OK);
    }
    break;
}

4. 和控件交互：读写编辑框文本

通过 SendMessageW 向控件发送系统消息，实现文本读取、修改，是Win32控件交互的核心方式。

  • WM_GETTEXTLENGTH：获取编辑框文本长度
  • WM_GETTEXT：读取编辑框文本到缓冲区
  • WM_SETTEXT：设置编辑框显示文本
#include <string>
// hEdit：编辑框HWND，输出std::wstring
std::wstring GetEditText(HWND hEdit)
{
    int len = (int)SendMessageW(hEdit, WM_GETTEXTLENGTH, 0, 0);
    if(len <=0) return L"";
    std::wstring buf(len+1,0);
    SendMessageW(hEdit, WM_GETTEXT, len+1, (LPARAM)buf.data());
    return buf;
}

消息区别：SendMessage 同步阻塞等待执行完毕；PostMessage 异步投递消息，立即返回，不阻塞线程。

5. GDI文字绘图 WM_PAINT

窗口遮挡、缩放、刷新时系统自动触发 WM_PAINT 消息，所有绘图逻辑必须写在此消息内，必须成对使用BeginPaint/EndPaint。

case WM_PAINT:
{
    PAINTSTRUCT ps;
    HDC hdc = BeginPaint(hwnd, &ps);

    //GDI绘制文字
    TextOutW(hdc, 20,100, L"Hello Win32 GDI", lstrlenW(L"Hello Win32 GDI"));

    EndPaint(hwnd, &ps);
    break;
}

6. 必须掌握的常见消息速查表

消息常量 触发时机
WM_CREATE 窗口创建完成，初始化子控件专用
WM_COMMAND 按钮、编辑框、菜单交互通知
WM_PAINT 窗口需要重绘，执行GDI绘图
WM_SIZE 窗口大小改变，手动适配控件布局
WM_DESTROY 窗口销毁，退出程序
WM_CLOSE 用户点击关闭按钮，可拦截确认弹窗

重点提醒：Win32 无自动布局！窗口拉伸后控件不会自适应，需捕获WM_SIZE消息，调用SetWindowPos()手动调整控件位置大小。

7. 典型踩坑（非小白必看）

  • A/W字符串混用：L""宽字符仅适配W-API，char*不可传入Unicode接口，直接乱码崩溃
  • 控件样式缺失：缺少WS_CHILD变独立窗口，缺少WS_VISIBLE控件隐藏
  • 消息循环错误：禁止while(true)死循环，必须判断GetMessage返回值
  • 句柄误区：HWND是系统内核句柄，非C++指针，不能delete释放
  • GDI资源泄漏：HDC、画笔、画刷必须手动DeleteObject释放
  • TCHAR宏不推荐：现代开发优先直接使用W后缀Unicode API，兼容性更强

9. 什么时候放弃 Win32 转 Qt

Win32适合底层原理学习，不适合复杂项目开发，出现以下情况建议转Qt：

  • 需要大量控件、复杂布局，手写适配成本过高
  • 需要表格、树形控件、富文本、动画等高级组件
  • 需要跨平台运行（Windows/Linux/macOS）

10. 经典标准版 Win32 窗口源码（ANSI 完整版）

全网通用教学模板，兼容所有编译器，带图标、边框、容错机制、结构清零，标准化工业写法。

#include <windows.h>

/* 窗口过程回调函数：操作系统推送所有窗口消息到此函数 */
LRESULT CALLBACK WndProc(HWND hwnd, UINT Message, WPARAM wParam, LPARAM lParam) {
	switch(Message) {
		/* 窗口销毁消息：用户点击关闭按钮触发 */
		case WM_DESTROY: {
			PostQuitMessage(0); // 投递退出消息，终止消息循环
			break;
		}
		/* 所有未自定义处理的消息，交由系统默认处理器处理（必须写） */
		default:
			return DefWindowProc(hwnd, Message, wParam, lParam);
	}
	return 0;
}

/* Win32 GUI 程序唯一入口，替代控制台 main */
int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow) {
	WNDCLASSEX wc;    // 窗口类属性结构体，定义窗口基础样式
	HWND hwnd;        // 窗口句柄，标识唯一窗口实例
	MSG msg;          // 消息结构体，存储系统推送的事件

	// 1. 结构体内存清零（规范写法，避免脏数据导致窗口注册失败）
	memset(&wc,0,sizeof(wc));

	// 2. 配置窗口类核心参数
	wc.cbSize		 = sizeof(WNDCLASSEX);        // 结构体大小，固定必填
	wc.lpfnWndProc	 = WndProc;                    // 绑定自定义消息回调函数
	wc.hInstance	 = hInstance;                  // 绑定当前程序实例句柄
	wc.hCursor		 = LoadCursor(NULL, IDC_ARROW);// 设置窗口默认鼠标指针

	wc.hbrBackground = (HBRUSH)(COLOR_WINDOW+1);  // 窗口背景色（系统默认白色）
	wc.lpszClassName = "WindowClass";              // 自定义窗口类名（全局唯一）
	wc.hIcon		 = LoadIcon(NULL, IDI_APPLICATION); // 窗口大图标
	wc.hIconSm		 = LoadIcon(NULL, IDI_APPLICATION); // 窗口小图标（任务栏）

	// 3. 注册窗口类到操作系统内核
	if(!RegisterClassEx(&wc)) {
		MessageBox(NULL, "Window Registration Failed!","Error!",MB_ICONEXCLAMATION|MB_OK);
		return 0;
	}

	// 4. 创建顶层窗口（带扩展样式）
	hwnd = CreateWindowEx(
		WS_EX_CLIENTEDGE,        // 扩展样式：窗口客户区凹陷边框
		"WindowClass",           // 绑定已注册的窗口类名
		"Caption",               // 窗口标题
		WS_VISIBLE|WS_OVERLAPPEDWINDOW, // 窗口样式：默认显示+标准窗口样式
		CW_USEDEFAULT,           // 窗口X坐标（系统默认）
		CW_USEDEFAULT,           // 窗口Y坐标（系统默认）
		640,                     // 窗口宽度
		480,                     // 窗口高度
		NULL,NULL,hInstance,NULL
	);

	// 窗口创建失败容错判断
	if(hwnd == NULL) {
		MessageBox(NULL,"Window Creation Failed!","Error!",MB_ICONEXCLAMATION|MB_OK);
		return 0;
	}

	// 消息循环
	while(GetMessage(&msg, NULL, 0, 0) > 0)
	{
		TranslateMessage(&msg);
		DispatchMessage(&msg);
	}
	return (int)msg.wParam;
}
